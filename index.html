const SPREADSHEET_NAME = 'SURA ROYAL LUXE - Job Applications'; const SHEET_NAME = 'Applications'; const DRIVE_FOLDER_NAME = 'SURA ROYAL LUXE Applicant Photos'; // প্রথমবার একবার Run করুন function setup() { const props = PropertiesService.getScriptProperties(); let spreadsheetId = props.getProperty('SPREADSHEET_ID'); let folderId = props.getProperty('PHOTO_FOLDER_ID'); let spreadsheet; if (spreadsheetId) { spreadsheet = SpreadsheetApp.openById(spreadsheetId); } else { spreadsheet = SpreadsheetApp.create(SPREADSHEET_NAME); spreadsheetId = spreadsheet.getId(); props.setProperty('SPREADSHEET_ID', spreadsheetId); } let sheet = spreadsheet.getSheetByName(SHEET_NAME); if (!sheet) sheet = spreadsheet.insertSheet(SHEET_NAME); if (sheet.getLastRow() === 0) { sheet.appendRow([ 'Application ID', 'Submitted At', 'Applicant Name', 'Mobile', 'Father Name', 'Emergency Contact', 'Mother Name', 'NID / Birth Registration', 'Date of Birth', 'Marital Status', 'Present Address', 'Permanent Address', 'Applied Position', 'Special Experience / Skills', 'Work Experience 1 - Company', 'Work Experience 1 - Position', 'Work Experience 1 - Duration', 'Work Experience 1 - Last Salary', 'Work Experience 2 - Company', 'Work Experience 2 - Position', 'Work Experience 2 - Duration', 'Work Experience 2 - Last Salary', 'Reference Name', 'Reference Mobile', 'Reference Relation', 'Reference NID', 'Photo URL' ]); sheet.setFrozenRows(1); sheet.getRange(1, 1, 1, sheet.getLastColumn()) .setFontWeight('bold'); } if (!folderId) { const folder = DriveApp.createFolder(DRIVE_FOLDER_NAME); folderId = folder.getId(); props.setProperty('PHOTO_FOLDER_ID', folderId); } else { DriveApp.getFolderById(folderId); } Logger.log('Spreadsheet URL: ' + spreadsheet.getUrl()); Logger.log( 'Photo folder URL: https://drive.google.com/drive/folders/' + folderId ); return { spreadsheetUrl: spreadsheet.getUrl(), photoFolderUrl: 'https://drive.google.com/drive/folders/' + folderId }; } // Web App খুললে HTML ফর্ম দেখাবে function doGet() { return HtmlService.createHtmlOutputFromFile('index') .setTitle('SURA ROYAL LUXE - Job Application Form'); } // ফর্মের তথ্য Sheet-এ জমা এবং ছবি Drive-এ সংরক্ষণ function submitApplication(data) { if (!data || typeof data !== 'object') { throw new Error('ফর্মের তথ্য পাওয়া যায়নি।'); } const props = PropertiesService.getScriptProperties(); const spreadsheetId = props.getProperty('SPREADSHEET_ID'); const folderId = props.getProperty('PHOTO_FOLDER_ID'); if (!spreadsheetId || !folderId) { throw new Error('প্রথমে setup() একবার চালান।'); } const lock = LockService.getScriptLock(); lock.waitLock(30000); try { const ss = SpreadsheetApp.openById(spreadsheetId); const sheet = ss.getSheetByName(SHEET_NAME); if (!sheet) { throw new Error('Applications sheet পাওয়া যায়নি। setup() চালান।'); } const now = new Date(); const applicationId = 'SRL-' + Utilities.formatDate( now, Session.getScriptTimeZone(), 'yyyyMMdd-HHmmss' ) + '-' + Math.floor(100 + Math.random() * 900); let photoUrl = ''; if (data.photoData && String(data.photoData).indexOf('data:image/') === 0) { const match = String(data.photoData).match( /^data:(image\/[a-zA-Z0-9.+-]+);base64,([\s\S]+)$/ ); if (!match) { throw new Error('ছবির ফরম্যাট সঠিক নয়। আবার ছবি নির্বাচন করুন।'); } const mimeType = match[1]; const bytes = Utilities.base64Decode(match[2]); if (bytes.length > 5 * 1024 * 1024) { throw new Error('ছবির সাইজ ৫ MB-এর কম রাখুন।'); } const safeName = String( data.photoName || 'applicant-photo' ).replace(/[\\/:*?"<>|]/g, '_'); const extension = mimeType.split('/')[1].replace('jpeg', 'jpg'); const blob = Utilities.newBlob( bytes, mimeType, applicationId + '-' + safeName.replace(/\.[^.]+$/, '') + '.' + extension ); const file = DriveApp.getFolderById(folderId).createFile(blob); // ছবিটি private থাকবে; তার URL Sheet-এ রাখা হবে photoUrl = file.getUrl(); } const exp = Array.isArray(data.workExperience) ? data.workExperience : []; const exp1 = Array.isArray(exp[0]) ? exp[0] : []; const exp2 = Array.isArray(exp[1]) ? exp[1] : []; const values = [ applicationId, now, data.fullName || '', data.phone || '', data.fatherName || '', data.emergencyPhone || '', data.motherName || '', data.nid || '', data.dob || '', data.maritalStatus || '', data.presentAddress || '', data.permanentAddress || '', data.appliedPosition || '', data.specialExperience || '', exp1[0] || '', exp1[1] || '', exp1[2] || '', exp1[3] || '', exp2[0] || '', exp2[1] || '', exp2[2] || '', exp2[3] || '', data.refName || '', data.refPhone || '', data.refRelation || '', data.refNid || '', photoUrl ]; sheet.appendRow(values); return { success: true, applicationId: applicationId, photoUrl: photoUrl, spreadsheetUrl: ss.getUrl() }; } finally { lock.releaseLock(); } }
